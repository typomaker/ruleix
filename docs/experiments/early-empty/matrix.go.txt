// M1 Max, Go 1.27.3, GOMAXPROCS=1, three interleaved 80,000x samples.
// 5f6e63e -> accepted: missing-customer mean 12.411 -> 6.323 us per
// 100-offer Local; mixed mean 31.889 -> 31.956 us. See the empty-path study.
func BenchmarkEmptyMatrix(b *testing.B) {
	rows, ids := loadProductionDumpBenchmark(b)
	index, err := ruleix.New[productionDumpConstraint, uuid.UUID](productionDumpSchema()).Build(ruleix.Zip(rows, ids))
	require.NoError(b, err)
	for _, mode := range []string{"zero", "storezero", "slotzero", "intersectzero", "alternating", "mixed", "tiny", "medium", "broad", "rotate"} {
		b.Run(mode, func(b *testing.B) {
			requests := productionDumpRequests(rows, 64)
			if mode == "rotate" {
				for _, request := range requests {
					original := slices.Clone(request)
					for i := range request {
						request[i] = original[i*37%len(original)]
					}
				}
			} else if mode != "mixed" {
				requests = productionDumpCohort(b, rows, ids, requests, mode)
			}
			samples := make([]time.Duration, 0, 100000)
			run := 0
			b.ReportAllocs()
			for b.Loop() {
				start := time.Now()
				searchProductionDumpRequest(index, requests[run%len(requests)], nil)
				samples = append(samples, time.Since(start))
				run++
			}
			slices.Sort(samples)
			b.ReportMetric(float64(samples[(99*len(samples)+99)/100-1].Nanoseconds()), "p99-ns/request")
			b.ReportMetric(float64(samples[(999*len(samples)+999)/1000-1].Nanoseconds()), "p99.9-ns/request")
		})
	}
}

// Same controls: Index 105.585 -> 106.046 us and warm Local 30.471 ->
// 30.501 us per 100 searches, with unchanged allocation traffic.
func BenchmarkEmptyPublicControls(b *testing.B) {
	rows, ids := loadProductionDumpBenchmark(b)
	index, err := ruleix.New[productionDumpConstraint, uuid.UUID](productionDumpSchema()).Build(ruleix.Zip(rows, ids))
	require.NoError(b, err)
	requests := productionDumpRequests(rows, 64)
	b.Run("Index", func(b *testing.B) {
		var result []uuid.UUID
		run := 0
		b.ReportAllocs()
		for b.Loop() {
			for _, query := range requests[run%len(requests)] {
				result = result[:0]
				index.Search(query, &result)
			}
			run++
		}
	})
	b.Run("WarmLocal", func(b *testing.B) {
		local := index.Local()
		defer local.Close()
		var result []uuid.UUID
		run := 0
		b.ReportAllocs()
		for b.Loop() {
			for _, query := range requests[run%len(requests)] {
				result = result[:0]
				local.Search(query, &result)
			}
			run++
		}
	})
}

// Same controls: one missing-customer offer 307.5 -> 256.7 ns;
// one mixed offer 2239 -> 2242 ns, with unchanged allocations.
func BenchmarkEmptyShort(b *testing.B) {
	rows, ids := loadProductionDumpBenchmark(b)
	index, err := ruleix.New[productionDumpConstraint, uuid.UUID](productionDumpSchema()).Build(ruleix.Zip(rows, ids))
	require.NoError(b, err)
	for _, mode := range []string{"zero", "mixed"} {
		requests := productionDumpRequests(rows, 64)
		if mode == "zero" {
			requests = productionDumpCohort(b, rows, ids, requests, mode)
		}
		b.Run(mode, func(b *testing.B) {
			for _, length := range []int{1, 4, 16} {
				b.Run(fmt.Sprint(length), func(b *testing.B) {
					turn := 0
					b.ReportAllocs()
					for b.Loop() {
						searchProductionDumpRequest(index, requests[turn%len(requests)][:length], nil)
						turn++
					}
				})
			}
		})
	}
}
