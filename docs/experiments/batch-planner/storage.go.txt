func TestPlannerLayout(t *testing.T) {
 rows,ids:=loadComparison(t)
 index,err:=ruleix.New[productionDumpConstraint,uuid.UUID](productionDumpSchema()).Build(ruleix.Zip(rows,ids));require.NoError(t,err)
 data,err:=json.MarshalIndent(index.PlannerLayout(),"","  ");require.NoError(t,err)
 require.NoError(t,os.WriteFile(os.Getenv("RULEIX_TRIAL_LAYOUT"),data,0600))
}

// Fixed 10x counts, interleaved twice per revision. Full 96,630-row index;
// input decoding/setup excluded. See docs/batch-planner.md for latest results.
func BenchmarkPlannerBuild(b *testing.B) {
 rows,ids:=loadComparison(b)
 builder:=ruleix.New[productionDumpConstraint,uuid.UUID](productionDumpSchema())
 values:=ruleix.Zip(rows,ids)
 b.ReportAllocs()
 for b.Loop(){index,err:=builder.Build(values);require.NoError(b,err);runtime.KeepAlive(index)}
}

func plannerHeap() uint64 {var memory runtime.MemStats;runtime.ReadMemStats(&memory);return memory.HeapAlloc}

// Keep every built index live across two GCs; caller/input data remain live.
func BenchmarkPlannerIndexRetained(b *testing.B) {
 rows,ids:=loadComparison(b)
 indexes:=make([]*ruleix.Index[productionDumpConstraint,uuid.UUID],b.N)
 builder:=ruleix.New[productionDumpConstraint,uuid.UUID](productionDumpSchema())
 values:=ruleix.Zip(rows,ids)
 runtime.GC();runtime.GC();before:=plannerHeap()
 b.ReportAllocs();b.ResetTimer()
 for i:=range indexes {index,err:=builder.Build(values);require.NoError(b,err);indexes[i]=index}
 b.StopTimer();runtime.GC();runtime.GC();after:=plannerHeap()
 runtime.KeepAlive(indexes);runtime.KeepAlive(values);runtime.KeepAlive(rows);runtime.KeepAlive(ids)
 b.ReportMetric(float64(int64(after)-int64(before))/float64(b.N),"retained-B/index")
}
