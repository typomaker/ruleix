import (
	"math/rand/v2"
	"testing"

func TestNEONBetweenMask(t *testing.T) {
	rng := rand.New(rand.NewPCG(13, 29))
	var data packedBetweenLanes
	for trial := range 100000 {
		for slot := range 16 {
			data.lower[slot] = byte(rng.Uint32())
			data.upper[slot] = byte(rng.Uint32())
			data.lowerWild[slot] = 0
			data.upperWild[slot] = 0
			if rng.IntN(5) == 0 {
				data.lowerWild[slot] = 255
			}
			if rng.IntN(5) == 0 {
				data.upperWild[slot] = 255
			}
		}
		lower, upper := uint64(rng.IntN(256)), uint64(rng.IntN(256))
		want := goBetweenMask(&data, lower, upper)
		if got := neonBetweenMask(&data, lower, upper); got != want {
			t.Fatalf("trial %d lower %d upper %d got %x want %x", trial, lower, upper, got, want)
		}
	}
}
// Apple M1 Max, Go 1.27.3, GOMAXPROCS=2, -benchtime=300ms,
// prepared sixteen-lane data: Go 22.17 ns/op, NEON 2.427 ns/op.
func BenchmarkNEONKernel(b *testing.B) {
	var data packedBetweenLanes
	for slot := range 16 {
		data.lower[slot] = byte(slot + 1)
		data.upper[slot] = byte(slot + 9)
	}
	b.Run("Go", func(b *testing.B) {
		var mask uint64
		for b.Loop() {
			mask ^= goBetweenMask(&data, 8, 12)
		}
		kernelSink = mask
	})
	b.Run("NEON", func(b *testing.B) {
		var mask uint64
		for b.Loop() {
			mask ^= neonBetweenMask(&data, 8, 12)
		}
		kernelSink = mask
	})
}

var kernelSink uint64

func TestNEONFusedAndSimpleMask(t *testing.T) {
	rng := rand.New(rand.NewPCG(31, 47))
	var data [2]packedBetweenLanes
	var pairs [8]uint32
	for trial := range 50000 {
		for lane := range 16 {
			for pred := range 2 {
				data[pred].lower[lane] = byte(rng.Uint32())
				data[pred].upper[lane] = byte(rng.Uint32())
				data[pred].lowerWild[lane] = 0
				data[pred].upperWild[lane] = 0
				if rng.IntN(7) == 0 {
					data[pred].lowerWild[lane] = 255
				}
				if rng.IntN(7) == 0 {
					data[pred].upperWild[lane] = 255
				}
			}
			pair := uint16(data[0].lower[lane]) | uint16(data[0].upper[lane])<<8
			shift := uint(16 * (lane % 2))
			pairs[lane/2] = pairs[lane/2]&^(uint32(65535)<<shift) | uint32(pair)<<shift
		}
		pos := [4]uint64{uint64(rng.IntN(256)), uint64(rng.IntN(256)), uint64(rng.IntN(256)), uint64(rng.IntN(256))}
		want := goBetweenMask(&data[0], pos[0], pos[1]) & goBetweenMask(&data[1], pos[2], pos[3])
		if got := neonFusedBetweenMask(&data, &pos); got != want {
			t.Fatalf("fused trial %d got %x want %x", trial, got, want)
		}
		data[0].lowerWild = [16]byte{}
		data[0].upperWild = [16]byte{}
		want = goBetweenMask(&data[0], pos[0], pos[1])
		if got := neonSimpleBetweenMask(&pairs, pos[0], pos[1]); got != want {
			t.Fatalf("simple trial %d got %x want %x", trial, got, want)
		}
	}
}
